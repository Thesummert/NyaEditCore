cmake_minimum_required(VERSION 3.21)
project(NyaEditCore VERSION 0.1.0 LANGUAGES CXX)
find_package(ICU 60 REQUIRED COMPONENTS uc i18n)
add_library(nya_core STATIC
    Encoding.cpp include/nya/core/Encoding.h FileIO.cpp include/nya/core/FileIO.h
    FileSearch.cpp include/nya/core/FileSearch.h Search.cpp include/nya/core/Search.h)
target_compile_features(nya_core PUBLIC cxx_std_17)
set_target_properties(nya_core PROPERTIES AUTOMOC OFF AUTORCC OFF CXX_EXTENSIONS OFF)
target_include_directories(nya_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(nya_core PUBLIC ICU::uc ICU::i18n)
if(MSVC)
    target_compile_options(nya_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(nya_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
